== 3. a table, and the names that reach it
> SELECT count(*) FROM taxi.zones
+--------+
|count(1)|
+--------+
|265     |
+--------+

> SELECT count(*) FROM spark_catalog.taxi.zones
+--------+
|count(1)|
+--------+
|265     |
+--------+

> USE taxi
> SELECT count(*) FROM zones
+--------+
|count(1)|
+--------+
|265     |
+--------+

> SELECT current_catalog(), current_database()
+-----------------+----------------+
|current_catalog()|current_schema()|
+-----------------+----------------+
|spark_catalog    |taxi            |
+-----------------+----------------+

== 4. a second catalog: SHOW CATALOGS lists it only after first use
> SHOW CATALOGS
+-------------+
|catalog      |
+-------------+
|spark_catalog|
+-------------+

> CREATE NAMESPACE IF NOT EXISTS h2.ops
> SHOW CATALOGS
+-------------+
|catalog      |
+-------------+
|h2           |
|spark_catalog|
+-------------+

> SHOW NAMESPACES IN h2
+------------------+
|namespace         |
+------------------+
|INFORMATION_SCHEMA|
|PUBLIC            |
|ops               |
+------------------+

> CREATE TABLE h2.ops.zone_owner (LocationID INT, owner STRING)
> INSERT INTO h2.ops.zone_owner VALUES (132, 'airport team'), (138, 'airport team'),
                                        (161, 'midtown team')
== 5. one query, two catalogs
> SELECT z.Zone, o.owner
     FROM spark_catalog.taxi.zones AS z JOIN h2.ops.zone_owner AS o USING (LocationID)
     ORDER BY z.Zone
+-----------------+------------+
|Zone             |owner       |
+-----------------+------------+
|JFK Airport      |airport team|
|LaGuardia Airport|airport team|
|Midtown Center   |midtown team|
+-----------------+------------+

== 6. USE a catalog, and unqualified names follow it

> SELECT current_catalog(), current_database()

|current_catalog()|current_schema()|

|h2               |ops             |
+-----------------+----------------+

> SHOW TABLES

|namespace|tableName |isTemporary|

|ops      |zone_owner|false      |
+---------+----------+-----------+

> USE spark_catalog.taxi
> SHOW TABLES
+---------+---------+-----------+
|namespace|tableName|isTemporary|
