== 1. INSERT OVERWRITE with no PARTITION clause, static mode
   partitionOverwriteMode = STATIC
> INSERT OVERWRITE TABLE by_day SELECT z.Borough AS borough, count(*) AS trips, to_date(t.tpep_pickup_datetime) AS day
     FROM trips_raw t JOIN zones z ON t.PULocationID = z.LocationID
     WHERE t.tpep_pickup_datetime >= '2026-06-02' AND t.tpep_pickup_datetime < '2026-06-03'
     GROUP BY z.Borough, to_date(t.tpep_pickup_datetime)
    day=2026-06-01  files before 1  after 0  -> deleted
    day=2026-06-02  files before 1  after 1  -> replaced
    day=2026-06-03  files before 1  after 0  -> deleted
